CREATE TABLE equipe_mec 
( 
 cpf VARCHAR(11) PRIMARY KEY,  
 data_admissao DATE NOT NULL,  
 telefone VARCHAR(15),  
 nome VARCHAR(150) NOT NULL
); 

CREATE TABLE servicos 
( 
 id_servico INT PRIMARY KEY AUTO_INCREMENT,  
 nome VARCHAR(100) NOT NULL,  
 descricao TEXT,  
 tempomed_execucao VARCHAR(50) NOT NULL
); 

CREATE TABLE cliente 
( 
 cpf VARCHAR(11) PRIMARY KEY,  
 nome VARCHAR(150) NOT NULL
); 

CREATE TABLE veiculo 
( 
 placa VARCHAR(10) PRIMARY KEY,  
 idcliente VARCHAR(11) NOT NULL,  
 modelo VARCHAR(100) NOT NULL,  
 ano VARCHAR(4) NOT NULL,  
 cor VARCHAR(30),  
 marca VARCHAR(100) NOT NULL,  
 nome_proprietario VARCHAR(150) NOT NULL,
 FOREIGN KEY (idcliente) REFERENCES cliente(cpf)
); 

CREATE TABLE realiza 
( 
 cpf VARCHAR(11),  
 id_servico INT,  
 PRIMARY KEY (cpf, id_servico),
 FOREIGN KEY (cpf) REFERENCES equipe_mec(cpf),
 FOREIGN KEY (id_servico) REFERENCES servicos(id_servico)
); 

CREATE TABLE ordem_servico 
( 
 id_ordem INT PRIMARY KEY AUTO_INCREMENT,
 placa VARCHAR(10) NOT NULL,  
 cpf VARCHAR(11) NOT NULL,  
 status VARCHAR(50) NOT NULL,  
 dataentrada DATE NOT NULL,  
 problema VARCHAR(255) NOT NULL,  
 valor DECIMAL(10,2),  
 datasaida DATE,
 FOREIGN KEY (placa) REFERENCES veiculo(placa),
 FOREIGN KEY (cpf) REFERENCES equipe_mec(cpf)
);
